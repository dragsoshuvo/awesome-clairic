const response = await env.AI.run("@cf/cloudflare/clairic", {
  model: "clairic",
  state: "Invoice INV-2291 from Northwind Supply, issued 2026-09-02, net 30.",
  fields: {
    invoice_number: { type: "text" },
    due_date: { type: "date", instructions: "Issue date plus the net payment days" },
    total: { type: "number", decimals: 2, unit: "USD" },
    days_to_pay: { type: "estimate", min: 0, max: 120, step: 1 },
  },
});

console.log(response.answers.due_date.value); // "2026-10-02"
